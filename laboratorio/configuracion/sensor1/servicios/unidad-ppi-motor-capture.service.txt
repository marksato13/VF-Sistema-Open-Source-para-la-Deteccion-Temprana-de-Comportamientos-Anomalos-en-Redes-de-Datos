[Unit]
Description=CyberFlow - buffer en anillo de PCAP para el motor
After=network-online.target suricata.service cyberflow-capture-nic.service
Wants=network-online.target
Requires=cyberflow-capture-nic.service

[Service]
Type=simple
# Arranca como root y suelta privilegios con -Z justo tras abrir el socket.
#
# SIN -W, a proposito. Con -G, la opcion -W hace que tcpdump SALGA con estado 0
# al completar esos N ficheros, y systemd lo relanza: medido en el sensor,
# NRestarts=1458 con reinicios cada 4m02s. Eso costaba dos cosas, las dos
# invisibles:
#   - RestartSec deja ~2 s sin capturar en cada relevo: 0,8 % del trafico que
#     nunca llega al anillo.
#   - tcpdump con -Z hace chown del PRIMER fichero de cada arranque a
#     tcpdump:tcpdump; los siguientes los crea ya sin privilegios y heredan el
#     grupo del directorio setgid. Ese primer fichero NO lo puede leer el
#     usuario del motor: 1 de cada 16, el 5,55 % de los bytes del anillo,
#     descartado en silencio.
# -W tampoco servia para podar: como el nombre lleva fecha, tcpdump no
# reutiliza ficheros. De eso se encarga cyberflow-limpieza.timer.
ExecStartPre=/usr/bin/install -d -o tcpdump -g m4rk -m 2750 /var/lib/ppi-motor-capture
ExecStart=/usr/bin/tcpdump -i ens37 -n -s 0 -U -B 65536 \
    -Z tcpdump -w /var/lib/ppi-motor-capture/live-%%Y%%m%%d%%H%%M%%S.pcap \
    -G 15
UMask=0027
Restart=always
RestartSec=2

[Install]
WantedBy=multi-user.target

# Cierre complementario de la 4A (4-nov-2026). Único cambio: tras arrancar tcpdump, devolver al
# grupo del directorio (m4rk) el primer fichero de ese arranque, que tcpdump deja tcpdump:tcpdump
# (ver /usr/local/sbin/cf-primer-pcap). No cambia ExecStart, ni -Z, ni UMask, ni el usuario.
# El «-» hace que un fallo del ajuste no pare la captura: se ve en el journal y en pcaps_ilegibles.
# Retirar: borrar este fichero, daemon-reload y restart (vuelve el comportamiento anterior).
[Service]
ExecStartPost=-/usr/local/sbin/cf-primer-pcap /var/lib/ppi-motor-capture m4rk
