# /etc/systemd/system/cyberflow-capture-nic.service
# Generado por scripts/setup/cyberflow_config.py desde configs/cyberflow.toml
# NO editar a mano: cualquier cambio se pierde en la siguiente generacion.
# Para cambiar algo, edita el .toml y vuelve a ejecutar el generador.

[Unit]
Description=CyberFlow - prepara ens37 para captura pasiva
After=network-online.target
Wants=network-online.target
Before=suricata.service

[Service]
Type=oneshot
RemainAfterExit=yes
# Sin promiscuo la vNIC filtra el espejo: ninguna trama va dirigida a su MAC.
ExecStart=/usr/sbin/ip link set ens37 up promisc on
# GRO/LRO/TSO/GSO recomponen segmentos y falsearian tamano de trama y tasas.
ExecStart=-/usr/sbin/ethtool -K ens37 gro off lro off tso off gso off
# Una sonda pasiva no debe emitir.
ExecStart=-/usr/sbin/sysctl -qw net.ipv6.conf.ens37.disable_ipv6=1
ExecStop=-/usr/sbin/ip link set ens37 promisc off

[Install]
WantedBy=multi-user.target
